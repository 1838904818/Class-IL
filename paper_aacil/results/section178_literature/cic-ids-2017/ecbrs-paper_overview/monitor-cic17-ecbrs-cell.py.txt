"""Bounded single-cell confirmation for the S188 CIC17 paper-memory request."""
import argparse
import ctypes
import hashlib
import json
import os
from pathlib import Path
import subprocess
import time
import psutil

p = argparse.ArgumentParser(description=__doc__)
p.add_argument('--seed',type=int,choices=range(10,15),required=True)
a = p.parse_args()
base = Path('@@EXPERIMENT_ROOT@@')
repo = Path('@@SOURCE@@')
python = Path('@@PYTHON_ENV@@/python.exe')
entry = 'ncd/run_section178_ecbrs_extended.py'
volume = ctypes.create_unicode_buffer(100)
assert ctypes.windll.kernel32.GetVolumeNameForVolumeMountPointW('@@RESULT_DRIVE@@/',volume,100)
assert volume.value.lower() == '\\\\?\\volume{d5ede140-354b-40b6-9bb9-8769e4828c62}\\'
assert psutil.disk_usage('@@RESULT_DRIVE@@/').free > 20*1024**3
assert psutil.virtual_memory().available > 12*1024**3
for process in psutil.process_iter(['pid','name','cmdline']):
    if process.info['pid'] == os.getpid() or process.info['name'] != 'python.exe':
        continue
    command = ' '.join(process.info['cmdline'] or [])
    assert not any(s in command for s in ['run_section123_fixed.py','run_section178_ecbrs','monitor-cic17','run-ecbrs-extended-smokes.py','run-cse-literature-smokes.py']), 'Another cell is active'
env = os.environ.copy()
env.update(TEMP=str(base/'tmp'),TMP=str(base/'tmp'),PYTHONDONTWRITEBYTECODE='1',OMP_NUM_THREADS='2',MKL_NUM_THREADS='2')
subprocess.run([str(python),'-I','-B','-u',str(base/'admin/check-ecbrs-extended-smoke.py'),
                str(base/'runs/section178/cic-ids-2017/ecbrs-paper-extended-smoke-seed10')],env=env,check=True)
out = base/f'runs/section178/cic-ids-2017/ecbrs-paper/seed{a.seed}'
assert not out.exists(), 'Existing result/checkpoints must not be overwritten or retrained'
files = sorted(set(repo.glob('ncd/**/*.py')) | set(repo.glob('ncd/configs/*.json')) |
               {repo/'src/methods/replay_dpmeans.py',repo/'paper_aacil/results/section123_schedule_selection.json'})
def hashes():
    return {f.relative_to(repo).as_posix():hashlib.sha256(f.read_bytes()).hexdigest() for f in files}
bound = hashes()
smoke_events = [json.loads(line) for line in (base/'logs/ecbrs-extended-smokes.stdout.log').read_text(encoding='utf-8').splitlines() if line.startswith('{')]
assert bound == next(row['source_sha256'] for row in smoke_events if row.get('event') == 'START')
assert any(row.get('event') == 'ALL_SMOKES_COMPLETE' for row in smoke_events)
if a.seed != 10:
    reference = [json.loads(line) for line in (base/'logs/ecbrs-paper-cic17-seed10.monitor.stdout.log').read_text(encoding='utf-8').splitlines() if line.startswith('{')]
    assert bound == next(row['source_sha256'] for row in reference if row.get('event') == 'START')
    assert any(row.get('event') == 'CELL_COMPLETED_PENDING_INDEPENDENT_METRIC_CHECK' for row in reference)
prefix = base/f'logs/ecbrs-paper-cic17-seed{a.seed}'
command = [str(python),'-I','-B','-u',entry,'--dataset','CIC-IDS-2017','--seed',str(a.seed),
           '--data-root',str(base/'data/derived/section21_clean_v1'),'--output-dir',str(out),'--log-prefix',str(prefix)]
print(json.dumps({'event':'START','command':command,'source_sha256':bound,
                  'git_commit':subprocess.check_output(['git','rev-parse','HEAD'],cwd=repo,text=True).strip(),
                  'work_order_commit':'a8f4c8615668d8f3fa22fedef55835cac96e1caf','work_order_section':188,
                  'max_wall_seconds':3300,'max_rss_bytes':10*1024**3,'min_available_memory_bytes':2*1024**3}),flush=True)
peak, samples, reason = 0, 0, None
started = time.monotonic()
with Path(str(prefix)+'.stdout.log').open('x',encoding='utf-8') as stdout, Path(str(prefix)+'.stderr.log').open('x',encoding='utf-8') as stderr:
    child = subprocess.Popen(command,cwd=repo,env=env,stdout=stdout,stderr=stderr,creationflags=subprocess.CREATE_NO_WINDOW)
    tracked = psutil.Process(child.pid)
    print(json.dumps({'event':'CHILD_STARTED','pid':child.pid}),flush=True)
    try:
        while child.poll() is None:
            try:
                rss = tracked.memory_info().rss
                cpu = tracked.cpu_times()
            except psutil.NoSuchProcess:
                break
            peak = max(peak,rss)
            elapsed = time.monotonic()-started
            available = psutil.virtual_memory().available
            if samples % 6 == 0:
                print(json.dumps({'event':'RESOURCE','pid':child.pid,'elapsed_seconds':elapsed,'rss_bytes':rss,
                                  'peak_rss_bytes':peak,'cpu_seconds':cpu.user+cpu.system,'system_available_bytes':available}),flush=True)
            if elapsed > 3300 or rss > 10*1024**3 or available < 2*1024**3:
                reason = 'Local budget exceeded; retain checkpoints and logs; no automatic retry'
                child.terminate()
                break
            samples += 1
            time.sleep(5)
    finally:
        if child.poll() is None:
            child.terminate()
        try:
            code = child.wait(timeout=30)
        except subprocess.TimeoutExpired:
            child.kill()
            code = child.wait()
elapsed = time.monotonic()-started
unchanged = hashes() == bound
print(json.dumps({'event':'TERMINAL','exit_code':code,'elapsed_seconds':elapsed,'sampled_peak_rss_bytes':peak,
                  'source_unchanged':unchanged,'reason':reason}),flush=True)
assert unchanged and code == 0 and reason is None
summary = json.loads((out/'summary.json').read_text(encoding='utf-8'))
assert summary['completed'] and summary['scientific_result'] and not summary['smoke']
print(json.dumps({'event':'CELL_COMPLETED_PENDING_INDEPENDENT_METRIC_CHECK','seed':a.seed,'output':str(out)}),flush=True)
